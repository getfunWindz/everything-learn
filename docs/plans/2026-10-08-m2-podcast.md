```python
def sniff_url(url: str):
    """URL 路由兜底：无 match 时依次尝试适配器的可选 sniff(url)（仅 http/https）"""
    if not (url or "").startswith(("http://", "https://")):
        return None
    for a in _ADAPTERS.values():
        sniff = getattr(a, "sniff", None)
        if callable(sniff):
            try:
                if sniff(url):
                    return a
            except Exception:
                continue
    return None
```

class PodcastAdapter:
    name = "podcast"; kinds = {"audio"}
    def match(self, url): ...        # /feed|/rss|.rss|.xml / feeds. / simplecast|megaphone|acast|blubrry|

    def fetch_meta(self, ref): ...   # Task 4
    def fetch_content(self, ref): ...# Task 5
    def fetch_media_url(self, ref, media="audio", lowest=False): ...  # Task 5

def _itunes_lookup_feed(podcast_id: str) -> str:
    d = requests.get("https://itunes.apple.com/lookup", params={"id": podcast_id}, timeout=30).json()
    feed = (d.get("results") or [{}])[0].get("feedUrl")
    if not feed: raise ValueError(f"Apple 查找失败：未找到 {podcast_id} 的 feedUrl")
    return feed

## 风险与对策

- **私有 feed（token 在 URL）**：原样使用用户给的 URL，不解析剥离查询参数。
- **长集数大文件**：Whisper 前已知时长（`itunes:duration`），沿用长视频 VAD 策略。
- **transcript 格式多样**：只支持 vtt/srt；其他类型（json/文本）忽略并走 Whisper。
- **Apple 查找 API 偶发限流**：失败给"改用 RSS 地址"的提示。
